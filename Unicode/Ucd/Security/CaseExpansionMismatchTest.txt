# CaseExpansionMismatchTest.txt
# UCD version: 16.0.0
# Security-spec version: 0.1.0
# Family: F4 — Case-mapping Expansion Mismatch (UAX #21)
# License: Apache-2.0
#
# Threat model: an attacker submits text whose case mapping
# under the default locale changes the codepoint count, breaking
# fixed-width buffers, length-based credential comparison, and
# database column bounds.  Classical shapes: ß → "SS" upper,
# ﬁ → "FI" upper, İ → "i + dot above" lower.
#
# Distinct from F3 `LocaleCaseInversion`, which fires on shapes
# whose case mapping diverges across locales.  F4 fires on shapes
# whose case mapping is locale-stable but length-changing under
# the default locale.
#
# Detection (priority order):
#   1. UpperExpansion (basePos, cp, expansionLen)
#   2. LowerExpansion (basePos, cp, expansionLen)
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the F4 detector:
#   Clear (= every cp case-maps to a single cp)
#   Hazard:UpperExpansion
#   Hazard:LowerExpansion

# @section Clear
# @level basic

# ASCII Hello — H → h and h → H, both 1-cp.  No expansion.
0048 0065 006C 006C 006F; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # Hello

# All-caps ABC — A → a, etc.  No expansion.
0041 0042 0043; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # ABC

# Han 中文 — no case mappings at all.
4E2D 6587; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # 中文

# Greek lowercase αβγ — Α/Β/Γ all 1-cp upper.
03B1 03B2 03B3; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # αβγ

# Cyrillic привет — 6 cps, all 1-cp upper.
041F 0440 0438 0432 0435 0442; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # привет

# Korean Hangul 한 — caseless.
D55C; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # 한

# All-lowercase abc — 1-cp upper.
0061 0062 0063; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # abc

# Arabic letters — caseless.
0627 0628 0629; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # ابة

# ASCII mixed case "HELLO world" — every cp is 1-cp under both
# directions of case mapping.
0048 0045 004C 004C 004F 0020 0077 006F 0072 006C 0064; Clear; ; upper_exp=0; lower_exp=0; max_exp=1; # HELLO world

# @section UpperExpansion
# @level basic

# ß alone — toUpper → "SS" (1 → 2).  The canonical sharp-s
# expansion attack.
00DF; Hazard:UpperExpansion; 0; upper_exp=1; lower_exp=0; max_exp=2; # ß → SS

# Latin small ligature fi (U+FB01) — toUpper → "FI" (1 → 2).
FB01; Hazard:UpperExpansion; 0; upper_exp=1; lower_exp=0; max_exp=2; # ﬁ → FI

# Latin small ligature ffi (U+FB03) — toUpper → "FFI" (1 → 3).
FB03; Hazard:UpperExpansion; 0; upper_exp=1; lower_exp=0; max_exp=3; # ﬃ → FFI

# "Straße" — six-cp German word with ß at position 4.  toUpper
# expansion fires at position 4.
0053 0074 0072 0061 00DF 0065; Hazard:UpperExpansion; 4; upper_exp=1; lower_exp=0; max_exp=2; # Straße

# Latin small ligature ff (U+FB00) — toUpper → "FF" (1 → 2).
FB00; Hazard:UpperExpansion; 0; upper_exp=1; lower_exp=0; max_exp=2; # ﬀ → FF

# Latin small ligature fl (U+FB02) — toUpper → "FL" (1 → 2).
FB02; Hazard:UpperExpansion; 0; upper_exp=1; lower_exp=0; max_exp=2; # ﬂ → FL

# Latin small ligature ffl (U+FB04) — toUpper → "FFL" (1 → 3).
FB04; Hazard:UpperExpansion; 0; upper_exp=1; lower_exp=0; max_exp=3; # ﬄ → FFL

# Greek letter iota with combining marks "ΐ" (U+0390) —
# toUpper → "Ϊ́" (1 → 3 cps, the precomposed iota with two
# combining marks decomposing to several uppercase forms).
0390; Hazard:UpperExpansion; 0; upper_exp=1; lower_exp=0; max_exp=3; # ΐ → 3 cps upper

# Word "Straße" repeated — two ß occurrences, both flagged via
# `upper_exp=2`, but the detector points at the FIRST.
0073 0074 0072 0061 00DF 0065 00DF; Hazard:UpperExpansion; 4; upper_exp=2; lower_exp=0; max_exp=2; # straße + ß

# @section LowerExpansion
# @level basic

# İ alone — toLower under default → "i + 0307" (1 → 2).  No
# upper expansion (İ stays İ), so the detector falls through
# to the lower scan.
0130; Hazard:LowerExpansion; 0; upper_exp=0; lower_exp=1; max_exp=2; # İ → i+0307

# aİa — İ at position 1.  Surrounding 'a' has no expansion in
# either direction, so the detector fires at position 1.
0061 0130 0061; Hazard:LowerExpansion; 1; upper_exp=0; lower_exp=1; max_exp=2; # aİa

# Word "Istanbul" with capital İ at position 0.
0130 0073 0074 0061 006E 0062 0075 006C; Hazard:LowerExpansion; 0; upper_exp=0; lower_exp=1; max_exp=2; # İstanbul

# Multi-İ — two capital-with-dot-above letters in a row.
0130 0130; Hazard:LowerExpansion; 0; upper_exp=0; lower_exp=2; max_exp=2; # İİ — two lower-expanders
