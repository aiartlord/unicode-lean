# Family: AdmissibilityFormDrift

#
# Threat model: a single input's whole-string admissibility
# verdict differs between the input and its NFKC form.  A
# pre-normalisation `isAllowedIdentifier` check reaches one
# verdict; a post-normalisation check on the same input reaches
# the other.  Attacker controls which check guards which stage.
#
# AdmissibilityFormDrift complements IdentifierFormDrift.
# IdentifierFormDrift is per-codepoint and compares
# Identifier_Status against the NFKD-head.
# AdmissibilityFormDrift is whole-string and compares
# isAllowedIdentifier across the NFKC boundary.  The two
# detectors are not redundant: decomposed Hangul jamos satisfy
# IdentifierFormDrift cleanly (identity NFKD, Restricted on both
# sides) but fire AdmissibilityFormDrift because the jamo
# sequence composes under NFKC into a precomposed Allowed Hangul
# syllable.
#
# Detection (single sub-threat, direction-agnostic):
#   AdmissibilityFormDrift (inputAdmissible, nfkcAdmissible) —
#     the two booleans differ.  No position is reported because
#     the predicate is whole-string.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the detector:
#   Clear (= same admissibility on both sides of NFKC)
#   Hazard:AdmissibilityFormDrift

# @section Clear
# @level basic

# ASCII admin — admissible on both sides, NFKC = input.
0061 0064 006D 0069 006E; Clear; ; input_adm=true; nfkc_adm=true; # admin

# ASCII Hello — XID_Start ✓, all Allowed, identity NFKC.
0048 0065 006C 006C 006F; Clear; ; input_adm=true; nfkc_adm=true; # Hello

# Precomposed Hangul 한 — Allowed and stable under NFKC.
D55C; Clear; ; input_adm=true; nfkc_adm=true; # 한 (precomposed)

# Greek lowercase αβγ — Allowed identifiers, identity NFKC.
03B1 03B2 03B3; Clear; ; input_adm=true; nfkc_adm=true; # αβγ

# ASCII letter + digit — admissible on both sides.
0078 0030; Clear; ; input_adm=true; nfkc_adm=true; # x0

# A leading digit — not a valid identifier on either side.
# Both isAllowedIdentifier calls return false, so the verdict
# is form-stable and this detector stays clear.
0030 0031 0032; Clear; ; input_adm=false; nfkc_adm=false; # 012 (leading-digit, both false)

# Cyrillic privet — admissible on both sides under UTS #39.
041F 0440 0438 0432 0435 0442; Clear; ; input_adm=true; nfkc_adm=true; # привет

# Plain Han text — admissible on both sides.
4E2D 6587; Clear; ; input_adm=true; nfkc_adm=true; # Han

# Arabic word — admissible identifier on both sides.
0641 0644 062F; Clear; ; input_adm=true; nfkc_adm=true; # Arabic

# Leading bidi control alone — not a valid identifier on either side.
202E; Clear; ; input_adm=false; nfkc_adm=false; # RLO alone (form-stable false)

# @section AdmissibilityFormDrift
# @level basic

# ﬁ ligature (U+FB01) — Restricted (input not admissible), NFKC
# decomposes to "fi" which is admissible.
FB01; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # ﬁ → fi

# Mathematical Italic "admin" — Math Italic letters are Restricted
# in IdentifierStatus.txt; NFKC folds them to ASCII admin.
1D44E 1D451 1D45A 1D456 1D45B; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # 𝑎𝑑𝑚𝑖𝑛

# Fullwidth Ａ — Restricted, NFKC folds to ASCII A.
FF21; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # Ａ

# Decomposed Hangul jamos for 한 — every jamo is Restricted
# (input not admissible), NFKC composes them to U+D55C which is
# Allowed.  This row is the canonical complementarity case
# between IdentifierFormDrift and AdmissibilityFormDrift:
# IdentifierFormDrift sees no per-cp shift (jamos have identity
# NFKD), AdmissibilityFormDrift sees the whole-string verdict flip.
1112 1161 11AB; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # decomposed 한

# Decomposed Hangul jamos for two syllables — same shape, larger
# input.  Composes under NFKC to 한국 (admissible).
1112 1161 11AB 1100 116E 11A8; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # decomposed 한국

# Roman numeral Ⅳ (U+2163) — Restricted, NFKC folds to "IV".
2163; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # Roman IV

# Math Italic "password" — eight-cp identifier folding to ASCII.
1D45D 1D44E 1D460 1D460 1D464 1D45C 1D45F 1D451; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # math italic password

# Fullwidth ｐａｓｓｗｏｒｄ — Restricted-fold to ASCII admin
# credential.
FF50 FF41 FF53 FF53 FF57 FF4F FF52 FF44; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # fullwidth password

# Circled digit ① — Restricted, NFKC = "1" Allowed (but digits
# alone aren't valid identifiers either, so this is a flip in
# the OTHER direction: input admissible? No — Restricted.  NFKC
# admissible? No — leading digit.  Both false, the detector
# stays clear).  To make this row a true AdmissibilityFormDrift
# case, prefix with an Allowed letter so NFKC = "a1" which IS a
# valid identifier.
0061 2460; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # a + circled-1 → a1

# Square HV (U+33CD) — Restricted, NFKC = "HV" admissible.
33CD; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # square HV

# Long compatibility chain — ﬃ ligature followed by Roman III.
FB03 2162; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # ﬃ + Roman III

# Math Fraktur "data" — four-cp math identifier folding to "data".
1D521 1D51E 1D531 1D51E; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # math fraktur data

# Halfwidth katakana 'ｱ' — Restricted compatibility halfwidth, NFKC
# folds to U+30A2 Allowed katakana.
FF71; Hazard:AdmissibilityFormDrift; ; input_adm=false; nfkc_adm=true; # halfwidth katakana
