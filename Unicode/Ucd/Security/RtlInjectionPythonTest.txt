# Companion fixture to RtlInjectionTokenizedTest.txt (the Rust
# grammar variant) that exercises D3's v1.5 region-aware
# behaviour under the Python tokenizer.  Every row is evaluated
# with `detect input .python`.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>

# @section Clear
# @level basic

# ASCII Hello — no payload, no bidi.
0048 0065 006C 006C 006F; Clear; ; # Hello

# RLO INSIDE a Python single-quote string.
0027 0041 202E 0042 0027; Clear; ; # 'A<RLO>B' filtered

# RLO INSIDE a Python double-quote string.
0022 0041 202E 0042 0022; Clear; ; # "A<RLO>B" filtered

# RLO INSIDE a Python `#` line comment.
0023 0020 202E; Clear; ; # # comment with RLO

# RLO INSIDE a Python triple-quoted string.
0022 0022 0022 0041 202E 0042 0022 0022 0022; Clear; ; # """A<RLO>B""" filtered

# Hebrew text INSIDE a Python string literal.
0022 05D0 05D1 05D2 0022; Clear; ; # Hebrew in string

# 5-char Hebrew run INSIDE a Python triple-quoted string.
0022 0022 0022 05D0 05D1 05D2 05D3 05D4 0022 0022 0022; Clear; ; # 5-Hebrew in triple-string

# @section Hazard
# @level basic

# RLO BEFORE a Python string literal.
202E 0022 0041 0022; Hazard:RloInLTRField; 0; # RLO + "A" in code prefix

# RLO at end of code AFTER a Python string closes.
0022 0048 0069 0022 202E; Hazard:RloInLTRField; 4; # "Hi" + trailing RLO in code

# Leading Hebrew BEFORE a Python string literal — field takeover.
05D0 0022 0041 0022; Hazard:FieldTakeover; 0; # Hebrew + string in code prefix

# Mid-stream Hebrew in pure code.
0041 0042 05D0 0044; Hazard:StrongRTLInLTR; 2; # AB + Hebrew + D in code

# 5-char Hebrew run in pure code.
0041 0042 05D0 05D1 05D2 05D3 05D4 0044; Hazard:MixedOverflow; 2; # AB + 5-Hebrew + D in code

# RLO BEFORE a `#` line comment.
202E 0023 0041; Hazard:RloInLTRField; 0; # RLO + # line comment

# RLE override in code followed by string containing Hebrew.
202B 0022 05D0 05D1 0022; Hazard:RloInLTRField; 0; # RLE + Hebrew in string
