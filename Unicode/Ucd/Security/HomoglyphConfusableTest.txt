# Threat model: identifiers whose visible glyph stream matches a
# canonical target's but whose byte stream differs by Cyrillic /
# Greek / Math-Alpha / Fullwidth / decomposition substitutions.
# Detects the Oct 2025 NuGet Nethereum typosquat class + IDN
# homograph + Math-Alpha posing + Fullwidth disguise.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the detector:
#   Clear
#   Hazard:TargetMatch          Hazard:MathAlpha
#   Hazard:WidthClass           Hazard:DecompositionSwap
#   Hazard:CrossScriptMix       Hazard:RestrictionLow
#
# Attribution conventions:
#   target="..."                matched canonical target name
#   first_math_cp=0xNNNN        first math-alpha codepoint
#   first_width_cp=0xNNNN       first fullwidth codepoint
#   script_count=N              distinct script families
#   restriction=LEVEL           UTS #39 restriction level

# @section Clear
# @level basic

# Plain ASCII "Hello" — no confusable pattern.
0048 0065 006C 006C 006F; Clear; ; restriction=ASCIIOnly; # ASCII Hello

# Legitimate "Nethereum" (pure Latin) — clear.
004E 0065 0074 0068 0065 0072 0065 0075 006D; Clear; ; restriction=ASCIIOnly; target="Nethereum"; # Legitimate Nethereum

# Legitimate "apple" — clear.
0061 0070 0070 006C 0065; Clear; ; restriction=ASCIIOnly; target="apple"; # Legitimate apple

# Legitimate "openai" — clear.
006F 0070 0065 006E 0061 0069; Clear; ; restriction=ASCIIOnly; target="openai"; # Legitimate openai

# Plain Han — single-script, no Latin lookalike — clear.
4E2D 6587; Clear; ; restriction=SingleScript; # Han 中文

# @section TargetMatch
# @level basic

# Nethereum Oct-2025 typosquat — final Cyrillic 'е' (U+0435) at position 6.
# Skeleton matches the canonical "Nethereum".
004E 0065 0074 0068 0065 0072 0435 0075 006D; Hazard:TargetMatch; ; target="Nethereum"; # Oct 2025 NuGet typosquat

# IDN homograph "apple" with Cyrillic 'а' (U+0430) at position 0.
0430 0070 0070 006C 0065; Hazard:TargetMatch; ; target="apple"; # IDN homograph apple, Cyrillic а

# IDN homograph "paypal" with Cyrillic 'а' (U+0430) at positions 1 and 4.
0070 0430 0079 0070 0430 006C; Hazard:TargetMatch; ; target="paypal"; # IDN homograph paypal

# @section DecompositionSwap
# @level basic

# e + combining acute — NFC = #[U+00E9].  Single decomposed letter,
# no Latin/non-Latin mix, no target.
0065 0301; Hazard:DecompositionSwap; 0; # e + combining acute → é

# o + combining diaeresis — NFC = #[U+00F6].
006F 0308; Hazard:DecompositionSwap; 0; # o + diaeresis → ö

# a + combining grave — NFC = #[U+00E0].
0061 0300; Hazard:DecompositionSwap; 0; # a + grave → à

# Two decomposed letters in sequence — NFC folds both.
0065 0301 0061 0300; Hazard:DecompositionSwap; 0; # decomposed é + à

# @section CrossScriptMix
# @level basic
#
# Two or more script families resolved in the same identifier
# without the Highly-Restrictive covered-script combination.  The
# union-side primitive `stringScriptUnion` answers "how many
# distinct script families appear here"; the historical
# intersection-based `stringResolvedScripts` was the wrong tool
# (Latin ∩ Cyrillic = ∅ ⇒ size 0).  CrossScriptMix fires whenever
# the union is ≥ 2 and the identifier is not Highly-Restrictive.
#
# Near-typosquats whose iterated skeleton doesn't quite fold to a
# canonical target also land here when the substitution dragged in
# a Cyrillic / Greek codepoint (microsoft / solana / metamask /
# binance — their iteratedSkeleton differs from the canonical by
# one bit, so TargetMatch doesn't fire, but the script-mix signal
# is unambiguous).

# "microsoft" with Cyrillic 'о' (U+043E) — Latin + Cyrillic mix.
006D 043E 0063 0072 006F 0073 006F 0066 0074; Hazard:CrossScriptMix; ; restriction=MinimallyRestrictive; # microsoft / Cyrillic о — near-typosquat

# "solana" with Cyrillic 'о' (U+043E).
0073 006F 043E 0061 006E 0061; Hazard:CrossScriptMix; ; restriction=MinimallyRestrictive; # solana near-typosquat

# "metamask" with Cyrillic 'а' (U+0430).
006D 0065 0430 0061 006D 0061 0073 006B; Hazard:CrossScriptMix; ; restriction=MinimallyRestrictive; # metamask near-typosquat

# "binance" with Cyrillic 'а' (U+0430).
0062 0069 0430 006E 0061 006E 0063 0065; Hazard:CrossScriptMix; ; restriction=MinimallyRestrictive; # binance near-typosquat

# Latin + Greek mix — `ab αβ`.
0061 0062 03B1 03B2; Hazard:CrossScriptMix; ; restriction=MinimallyRestrictive; # Latin+Greek mix

# Latin + Cyrillic mix — `xy фф`.
0078 0079 0444 0444; Hazard:CrossScriptMix; ; restriction=MinimallyRestrictive; # Latin+Cyrillic mix

# Greek + Cyrillic mix — αβ + Cyrillic ф х.
03B1 03B2 0444 0445; Hazard:CrossScriptMix; ; restriction=MinimallyRestrictive; # Greek+Cyrillic mix

# @section RestrictionLow
# @level basic
#
# Single-script inputs whose UTS #39 Identifier_Status is
# Restricted for at least one codepoint, pushing the restriction
# level to Unrestricted.  CrossScriptMix doesn't fire because the
# script union is size 1; the hazard is the restricted-status
# coverage of obscure / historical scripts.

# Ahom letter A (U+11700) — Restricted-status historical script.
11700; Hazard:RestrictionLow; ; restriction=Unrestricted; # Ahom letter A — historical script

# Cuneiform sign A (U+12000) — Restricted-status.
12000; Hazard:RestrictionLow; ; restriction=Unrestricted; # Cuneiform sign A

# Egyptian Hieroglyph A001 (U+13000) — Restricted-status.
13000; Hazard:RestrictionLow; ; restriction=Unrestricted; # Egyptian Hieroglyph A001
