# WidthClassConfusionTest.txt
# UCD version: 16.0.0
# Security-spec version: 0.1.0
# Family: F5 — Width Class Confusion (UAX #11)
# License: Apache-2.0
#
# Threat model: an attacker submits text containing Fullwidth
# (EAW = F) or Halfwidth (EAW = H) codepoints whose NFKD fold
# changes the East Asian Width class.  The canonical attack:
# `ＡＤＭＩＮ` (U+FF21..FF26) passes an ASCII-whitelist validator
# (or fails it but evades the check), then NFKC-normalises to
# `ADMIN` at storage / comparison, yielding two-system bypass.
#
# Distinct from D4 `FullwidthVariance`, which fires on the same
# input shapes for renderer-cohort divergence reasons.  Both
# detectors can fire on the same input — that is the layered
# model.  Pure Hangul stays clear because its NFKD output jamos
# are themselves EAW = W (no class change).
#
# Detection (priority order, both reachable):
#   1. FullwidthFold (basePos, cp, foldedCp) — EAW = F → ≠ F
#   2. HalfwidthFold (basePos, cp, foldedCp) — EAW = H → ≠ H
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the F5 detector:
#   Clear (= no width-class fold)
#   Hazard:FullwidthFold
#   Hazard:HalfwidthFold

# @section Clear
# @level basic

# ASCII Hello — every cp is EAW = Na, identity NFKD.
0048 0065 006C 006C 006F; Clear; ; fw_fold=0; hw_fold=0; # Hello

# Capital ASCII — EAW = Na.
0041 0044 004D 0049 004E; Clear; ; fw_fold=0; hw_fold=0; # ADMIN (plain ASCII)

# Plain Han 中文 — EAW = W with identity NFKD.
4E2D 6587; Clear; ; fw_fold=0; hw_fold=0; # 中文

# Korean Hangul 한 — EAW = W; NFKD = three jamos, all EAW = W,
# so no width-class change.  Pure Hangul must stay clear.
D55C; Clear; ; fw_fold=0; hw_fold=0; # 한

# Korean 한글 — two syllables, both jamos-of-jamos still W.
D55C AE00; Clear; ; fw_fold=0; hw_fold=0; # 한글

# Plain Cyrillic — every cp is EAW = N, identity NFKD.
041F 0440 0438 0432 0435 0442; Clear; ; fw_fold=0; hw_fold=0; # привет

# Plain Hiragana (EAW = W) — identity NFKD.
3042 3044 3046; Clear; ; fw_fold=0; hw_fold=0; # あいう

# Plain fullwidth katakana ア (EAW = W, NOT EAW = F).
30A2 30A4 30A6; Clear; ; fw_fold=0; hw_fold=0; # アイウ (W not F)

# Greek lowercase — EAW = N, identity NFKD.
03B1 03B2 03B3; Clear; ; fw_fold=0; hw_fold=0; # αβγ

# @section FullwidthFold
# @level basic

# Fullwidth Ａ (U+FF21) — EAW = F, NFKD = U+0041 (EAW = Na).
# The canonical Latin-impersonation attack at position 0.
FF21; Hazard:FullwidthFold; 0; fw_fold=1; hw_fold=0; # Ａ → A

# Fullwidth ＡＤＭＩＮ — first fold at position 0.
FF21 FF24 FF2D FF29 FF2E; Hazard:FullwidthFold; 0; fw_fold=5; hw_fold=0; # ＡＤＭＩＮ → ADMIN

# Fullwidth digits ０１２３ — NFKD to ASCII digits.
FF10 FF11 FF12 FF13; Hazard:FullwidthFold; 0; fw_fold=4; hw_fold=0; # ０１２３ → 0123

# Mixed ASCII + fullwidth letter at position 4.
0041 0042 0043 0044 FF25; Hazard:FullwidthFold; 4; fw_fold=1; hw_fold=0; # ABCDＥ

# Fullwidth lowercase ｐａｓｓｗｏｒｄ — the validator-bypass shape
# for credential checks.  Folds via NFKC to "password".
FF50 FF41 FF53 FF53 FF57 FF4F FF52 FF44; Hazard:FullwidthFold; 0; fw_fold=8; hw_fold=0; # ｐａｓｓｗｏｒｄ → password

# Fullwidth punctuation, e.g. ！ (U+FF01) → ASCII !
FF01; Hazard:FullwidthFold; 0; fw_fold=1; hw_fold=0; # ！ → !

# Fullwidth slash ／ (U+FF0F) → /
FF0F; Hazard:FullwidthFold; 0; fw_fold=1; hw_fold=0; # ／ → /

# @section HalfwidthFold
# @level basic

# Halfwidth katakana ｱ (U+FF71) — EAW = H, NFKD = U+30A2
# (EAW = W).  The halfwidth katakana attack at position 0.
FF71; Hazard:HalfwidthFold; 0; fw_fold=0; hw_fold=1; # ｱ → ア

# Halfwidth ｲｳｴ (FF72..FF74) sequence — all H → W.
FF72 FF73 FF74; Hazard:HalfwidthFold; 0; fw_fold=0; hw_fold=3; # ｲｳｴ → イウエ

# Halfwidth katakana following a fullwidth katakana — fullwidth
# variant doesn't fire (no F → ≠F fold from W input), so the
# detector falls through to the halfwidth scan.  H position 1.
30A2 FF72; Hazard:HalfwidthFold; 1; fw_fold=0; hw_fold=1; # アｲ

# Halfwidth Hangul ｦ (FF66) — NFKD to a wide jamo.
FF66; Hazard:HalfwidthFold; 0; fw_fold=0; hw_fold=1; # ｦ halfwidth

# Halfwidth Hangul filler ﾞ (FF9E) — voicing mark.
FF9E; Hazard:HalfwidthFold; 0; fw_fold=0; hw_fold=1; # ﾞ voicing mark

# Mixed halfwidth in ASCII context.
0041 FF71 0042; Hazard:HalfwidthFold; 1; fw_fold=0; hw_fold=1; # AｱB
