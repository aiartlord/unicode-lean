# Family: F2 — Stream-Safe Text Format violation (UAX #15 § 13)
# License: Apache-2.0
#
# Threat model: an attacker submits an input with a long
# combining-mark run ("Zalgo" shape) that forces an unbounded
# normalization buffer at the receiver.  UAX #15 § 13 caps the
# run length at 30 non-starters and prescribes U+034F COMBINING
# GRAPHEME JOINER as the canonical breaker.  F2 flags inputs
# whose maximum non-starter run exceeds the cap.
#
# Detection (single sub-threat in v1):
#   * StreamSafeOverrun (basePos, runLen) — first contiguous
#     non-starter run with length > 30.
#
# Distinct from D4 `combiningStackOverflow`, which fires at the
# cosmetic Zalgo threshold of 4 marks per base for renderer-
# divergence reasons.  F2 is the spec-mandated DoS bound.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the F2 detector:
#   Clear (= within the Stream-Safe bound)
#   Hazard:StreamSafeOverrun

# @section Clear
# @level basic

# ASCII Hello — zero non-starters.
0048 0065 006C 006C 006F; Clear; ; max_run=0; overruns=0; total_ns=0; # Hello

# Korean Hangul 한 — a precomposed starter, zero non-starters.
D55C; Clear; ; max_run=0; overruns=0; total_ns=0; # 한

# Single combining mark after a base — within bound.
