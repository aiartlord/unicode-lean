# CovertDisplayCompoundTest.txt
# UCD version: 16.0.0
# Security-spec version: 0.1.0
# Family: X2 — Cross-Layer Covert × Display Compound (UAX #9 + UTS #51)
# License: Apache-2.0
#
# Threat model: a single input simultaneously carries a Layer-1
# covert channel (unregistered VS bytes per C2, or tag-block
# characters per C4) AND a Layer-3 display-deception channel
# (UAX #9 bidi format controls per D3).  The compound is
# strictly worse than the sum: C2/C4 alone catch the payload
# but miss the visual deception; D3 alone catches the bidi but
# misses the payload.  X2 catches the simultaneity.
#
# Detection (priority order, both reachable):
#   1. BidiPlusUnregisteredVs (bidiPos, vsPos) — bidi format
#        control + a VS occurrence whose classifyVSPosition is
#        `.suspicious` (no row in StandardizedVariants.txt or
#        emoji-variation-sequences.txt).
#   2. BidiPlusTagBlock (bidiPos, tagPos) — bidi format
#        control + a tag-block character (U+E0000..U+E007F).
#        Reached when no unregistered VS fires first.
#
# Bidi-only inputs stay clear under X2 (D3 covers them).
# VS-only inputs stay clear under X2 (C2 covers them).
# Standardized VS sequences (e.g. emoji + FE0F) do not fire even
# when paired with a bidi control because they are registered.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the X2 detector:
#   Clear (= no compound)
#   Hazard:BidiPlusUnregisteredVs
#   Hazard:BidiPlusTagBlock

# @section Clear
# @level basic

# ASCII Hello — no bidi, no VS, no tag block.
0048 0065 006C 006C 006F; Clear; ; # Hello

# Bidi-only — RLO alone is a D3 case, not an X2 case.
202E; Clear; ; # RLO alone

# VS-only — ASCII A + VS1 is a C2 case (suspicious VS), but no
# bidi control so X2 stays clear.
0041 FE00; Clear; ; # A + VS1 (C2 territory, not X2)

# Bidi + registered VS — RLO followed by emoji 😀 + FE0F.  The
# VS is sanctioned (emoji-variation-sequences.txt row), so
# classifyVSPosition is `.registeredEmojiPresentation`, NOT
# `.suspicious`.  X2 stays clear; D3 alone handles the bidi.
202E 1F600 FE0F; Clear; ; # RLO + grinning face + VS16 (registered)

# Bidi + bidi only — open-and-close RLO/PDF without payload.
202E 0041 0042 202C; Clear; ; # RLO AB PDF

# Plain Han text — no bidi, no VS, no tag block.
4E2D 6587; Clear; ; # 中文

# Tag-block character alone — no bidi, so X2 stays clear (C4
# handles the tag-block surface).
0041 E0001; Clear; ; # A + tag block (C4 territory)

# Bidi + tag-block-language-tag-END (U+E007F) alone — only one
# tag character; need both bidi AND tag block for X2 to fire.
# Wait, this DOES fire — that's the point of the compound.  Let
# me give the legitimate clear case instead: balanced LRE + PDF
# with no compound elements.
202A 0041 202C; Clear; ; # LRE A PDF (D3 territory)

# @section BidiPlusUnregisteredVs
# @level strict

# RLO + ASCII A + VS1 — the canonical compound.  Bidi at 0,
# unregistered VS at 2 (A + FE00 isn't in StandardizedVariants).
202E 0041 FE00; Hazard:BidiPlusUnregisteredVs; 0,2; # RLO A + VS1

# Visible "admin" + RLO + ASCII X + VS1 — bidi at 5, unreg VS
# at 7.  The attacker hides the reorder mid-string.
0061 0064 006D 0069 006E 202E 0058 FE00; Hazard:BidiPlusUnregisteredVs; 5,7; # admin + RLO + X + VS1

# LRI isolate + ASCII + VS3 — isolate-class bidi control also
# fires when paired with unregistered VS.
2067 0042 FE02; Hazard:BidiPlusUnregisteredVs; 0,2; # LRI + B + VS3

# LRE embedding + Han + unregistered VS — embedding-class bidi
# control variant.  Han is not in StandardizedVariants for VS5.
202A 4E2D FE05; Hazard:BidiPlusUnregisteredVs; 0,2; # LRE + Han + VS5

# FSI first-strong isolate + ASCII + supplementary VS17.
2068 0041 E0100; Hazard:BidiPlusUnregisteredVs; 0,2; # FSI + A + VS17

# RLE override + Cyrillic + VS6.
202B 0430 FE06; Hazard:BidiPlusUnregisteredVs; 0,2; # RLE + Cyrillic а + VS6

# @section BidiPlusTagBlock
# @level strict

# RLO + ASCII + tag block U+E0001 LANGUAGE TAG.  No suspicious
# VS in this input, so the detector falls through past the VS
# scan and fires the tag-block sub-threat.
202E 0041 E0001; Hazard:BidiPlusTagBlock; 0,2; # RLO A LANGUAGE_TAG

# Visible filename + RLO + tag block — bidi at 5, tag at 6.
0066 006F 006F 002E 0074 202E E0041; Hazard:BidiPlusTagBlock; 5,6; # foo.t + RLO + tag A

# RLO + registered emoji + tag block — the registered VS does
# NOT fire BidiPlusUnregisteredVs (it's sanctioned), but the
# tag block does fire BidiPlusTagBlock.  Validates the priority
# fall-through.
202E 1F600 FE0F E0042; Hazard:BidiPlusTagBlock; 0,3; # RLO + 😀 + VS16 + tag B

# LRI + ASCII + tag block.
2067 0042 E0001; Hazard:BidiPlusTagBlock; 0,2; # LRI + B + LANGUAGE TAG

# FSI + tag-block ETX (E0003).
2068 E0003; Hazard:BidiPlusTagBlock; 0,1; # FSI + tag E0003

# LRE + Han + tag-block.
202A 4E2D E0021; Hazard:BidiPlusTagBlock; 0,2; # LRE + Han + tag !

# RLE override + Han + tag-block — embedding-class bidi pairing.
202B 4E2D E0041; Hazard:BidiPlusTagBlock; 0,2; # RLE + Han + tag A
