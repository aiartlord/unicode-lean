# AiWatermarkDetectabilityTest.txt
# UCD version: 16.0.0
# Security-spec version: 0.1.0
# Family: AiWatermarkDetectability (character-level)
# License: Apache-2.0
#
# Threat model: an input may carry character-level markers
# consistent with a known AI provider's watermark scheme.
# Two adversarial shapes are in scope:
#
#   * Provenance-preserving watermark — invisible markers
#     (NNBSP at word boundaries, Variation Selectors in plain
#     text, default-ignorable carriers) injected for
#     downstream attribution.  An attacker may attempt
#     removal via normalisation.
#   * Adversarial injection — fake markers consistent with
#     provider P's scheme injected to discredit human-written
#     text as AI-generated.
#
# Detection priority (first match wins):
#   1.  Adversarial             — NNBSP arithmetic ≥ 3 markers
#   2.  Gpt5ZwspModulo          — ZWSP arithmetic ≥ 3 markers
#   3.  Unknown                 — ≥ 2 distinct invisible categories
#   4.  NnbspBoundary           — single-category NNBSP
#   5.  VariationSelectorCarrier — VS NOT adjacent to emoji
#   6.  ZwjNonEmoji              — ZWJ NOT adjacent to emoji
#   7.  SmartQuoteAlternation   — paired curly quotes, no straights
#   8.  EmDashPattern           — em-dashes, no hyphen-minus
#   9.  StatisticalTokenChoice  — input contains AI-favored vocab
#   10. DefaultIgnorableCarrier — single-category residual DI
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Attribution keys recognised by the AiWatermarkDetectability
# conformance harness:
#   markerCount=<N>  — codepoint count of the fired scheme's markers
#                      (0 when clear)
#
# Classifications produced by the AiWatermarkDetectability detector:
#   Clear
#   Hazard:NnbspBoundary
#   Hazard:VariationSelectorCarrier
#   Hazard:ZwjNonEmoji
#   Hazard:DefaultIgnorableCarrier

# @section Clear
# @level basic

# Empty input — vacuous no-watermark.
; Clear; ; markerCount=0; # empty input

# Pure ASCII "abc" — no markers.
61 62 63; Clear; ; markerCount=0; # ASCII "abc"

# Han "中文" — CJK content with no marker codepoints.
4E2D 6587; Clear; ; markerCount=0; # 中文

# Pure emoji 😀 — emoji on its own is not a watermark.
1F600; Clear; ; markerCount=0; # 😀 alone

# Emoji-ZWJ sequence 👩‍🔬 (woman + ZWJ + microscope = woman-
# scientist).  The ZWJ is adjacent to emoji bases on BOTH
# sides — legitimate emoji-ZWJ sequence, the detector stays clear.
1F469 200D 1F52C; Clear; ; markerCount=0; # 👩‍🔬 emoji-ZWJ sequence

# @section Clear
# @level strict

# Emoji + VS16 — explicit emoji-presentation selector after
# an emoji base.  Legitimate UTS #51 usage; the detector stays clear.
# Pins the VS emoji-adjacency exclusion.
1F600 FE0F; Clear; ; markerCount=0; # 😀 + VS16 emoji presentation

# @section NnbspBoundary
# @level basic

# Single U+202F NNBSP between ASCII letters.
61 202F 62; Hazard:NnbspBoundary; 1; markerCount=1; # "a NNBSP b"

# Two NNBSPs at distinct positions — aggregated under a
# single NnbspBoundary verdict, positions list both.
61 202F 62 202F 63; Hazard:NnbspBoundary; 1, 3; markerCount=2; # "a NNBSP b NNBSP c"

# @section VariationSelectorCarrier
# @level basic

# VS1 (U+FE00) in plain text — not adjacent to any emoji.
61 FE00 62; Hazard:VariationSelectorCarrier; 1; markerCount=1; # "a VS1 b"

# VS16 (U+FE0F) in plain text — NOT after an emoji, so the
# emoji-adjacency exclusion does not apply; fires.
61 FE0F 62; Hazard:VariationSelectorCarrier; 1; markerCount=1; # "a VS16 b" plain text

# Plane-14 IVS1 (U+E0100) in plain text — same shape, fires.
61 E0100 62; Hazard:VariationSelectorCarrier; 1; markerCount=1; # "a IVS1 b"

# @section ZwjNonEmoji
# @level basic

# Single ZWJ between ASCII letters — not adjacent to any
# emoji.  Fires ZwjNonEmoji.
61 200D 62; Hazard:ZwjNonEmoji; 1; markerCount=1; # "a ZWJ b"

# @section DefaultIgnorableCarrier
# @level basic

# SOFT HYPHEN (U+00AD) — default-ignorable, not classified by
# the three higher-priority probes.  Fires the catch-all.
61 00AD 62; Hazard:DefaultIgnorableCarrier; 1; markerCount=1; # "a SOFT-HYPHEN b"

# ZWSP (U+200B) — default-ignorable, not ZWJ, not VS, not
# NNBSP.  Fires DefaultIgnorableCarrier.  ZeroWidthPayload's
# BareZeroWidth also fires on the same byte (covert-channel
# lens); AiWatermarkDetectability is
# the watermark-attribution lens.  Composite rejection at
# higher Levels.
61 200B 62; Hazard:DefaultIgnorableCarrier; 1; markerCount=1; # "a ZWSP b"

# COMBINING GRAPHEME JOINER (U+034F) — default-ignorable
# combining mark.  Fires DefaultIgnorableCarrier.
61 034F 62; Hazard:DefaultIgnorableCarrier; 1; markerCount=1; # "a CGJ b"

# @section Unknown
# @level basic

# Multi-category mixing: NNBSP + SOFT HYPHEN co-occur in the
# same input.  Cannot attribute to a single scheme; fires
# Unknown (priority 3).  Positions list both invisible
# codepoints; markerCount = 2 (NNBSP count 1 + DI count 1).
61 202F 00AD 62; Hazard:Unknown; 1, 2; markerCount=2; # NNBSP + DI: two categories mixed

# Multi-category mixing: VS-non-emoji + ZWJ-non-emoji.
61 FE0F 200D 62; Hazard:Unknown; 1, 2; markerCount=2; # VS + ZWJ: two categories mixed

# @section Unknown
# @level strict

# Three-category mixing: NNBSP + VS + DI.  markerCount = 3.
61 202F 62 FE0F 63 00AD 64; Hazard:Unknown; 1, 3, 5; markerCount=3; # NNBSP + VS + DI: three categories

# @section Adversarial
# @level basic

# Three NNBSPs at arithmetic-progression positions (1, 3, 5 —
# gap 2) — too-regular placement, fires Adversarial.  This is
# the v1 character-level heuristic for "fake watermark
# injection" per the L6 spec's invariant 3.
61 202F 62 202F 63 202F 64; Hazard:Adversarial; 1, 3, 5; markerCount=3; # NNBSP every 2 cps — too regular

# @section Gpt5ZwspModulo
# @level basic

# Three ZWSPs at arithmetic-progression positions (1, 3, 5).
# Fires Gpt5ZwspModulo — the hypothetical GPT-5 per-provider
# modulo schedule, character-level heuristic.
61 200B 62 200B 63 200B 64; Hazard:Gpt5ZwspModulo; 1, 3, 5; markerCount=3; # ZWSP every 2 cps — modulo pattern

# @section SmartQuoteAlternation
# @level basic

# Paired curly quotes (U+201C ... U+201D) wrapping ASCII
# content, with NO ASCII straight quotes.  Stylistic AI
# signature: "AI prefers typographic quotes."
201C 61 62 63 201D; Hazard:SmartQuoteAlternation; 0, 4; markerCount=2; # "abc" with curly double quotes

# Paired single curly quotes (U+2018 ... U+2019).
2018 78 79 2019; Hazard:SmartQuoteAlternation; 0, 3; markerCount=2; # 'xy' with curly single quotes

# @section EmDashPattern
# @level basic

# Two em-dashes (U+2014) with no ASCII hyphen-minus.
# Stylistic AI signature: "AI prefers em-dashes."
# Input: "ab — cd — ef" using em-dash + spaces.
61 62 20 2014 20 63 64 20 2014 20 65 66; Hazard:EmDashPattern; 3, 8; markerCount=2; # two em-dashes, no hyphen

# @section StatisticalTokenChoice
# @level basic

# Input contains the AI-favored word "delve" — fires
# statisticalTokenChoice at position 0.
64 65 6C 76 65; Hazard:StatisticalTokenChoice; 0; markerCount=1; # the word "delve" alone

# "moreover" embedded inside a larger ASCII string.
3B 20 6D 6F 72 65 6F 76 65 72 2C 20; Hazard:StatisticalTokenChoice; 2; markerCount=1; # "; moreover, "

# @section Adversarial
# @level strict

# Priority pin: Adversarial outranks NnbspBoundary when the
# NNBSP count >= 3 AND positions are arithmetic.  Same input
# as the basic Adversarial row, repeated here under strict to
# document the priority decision point.
61 202F 62 202F 63 202F 64; Hazard:Adversarial; 1, 3, 5; markerCount=3; # priority: Adversarial outranks NnbspBoundary

# @section Gpt5ZwspModulo
# @level strict

# Priority pin: Gpt5ZwspModulo outranks DefaultIgnorableCarrier
# when the ZWSP count >= 3 AND positions are arithmetic.  Same
# input as the basic row, repeated here under strict to
# document the priority.
61 200B 62 200B 63 200B 64; Hazard:Gpt5ZwspModulo; 1, 3, 5; markerCount=3; # priority: Gpt5ZwspModulo outranks DI
