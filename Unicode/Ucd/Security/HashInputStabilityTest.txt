# HashInputStabilityTest.txt
# UCD version: 16.0.0
# Security-spec version: 0.1.0
# Family: HashInputStability (UAX #15 + RFC 8785)
# License: Apache-2.0
#
# Threat model: an input hashed by a signer must be byte-identical
# to the input hashed by the verifier.  If the two ends pick
# different canonical forms (NFC vs NFD, trim convention, line
# ending) the hashes diverge silently — yet both sides believe
# they signed the same content.  Concrete pipelines:
#
#   * PGP signed messages (RFC 4880 / 9580)
#   * RFC 8785 JSON canonicalization (NFC requirement)
#   * Audit-log entries (line-ending normalisation on disk write)
#   * Webhook signatures (client vs server byte conventions)
#
# Canonical form:
#   hashStable input = trimTrailing (NFC input)
#
# `trimTrailing` strips ASCII whitespace only — U+0020, U+0009,
# U+000A, U+000D.  Unicode whitespace (U+00A0 NBSP, U+3000
# IDEOGRAPHIC SPACE) is content, not framing.
#
# Detection priority (first hit wins):
#   1. TrailingWhitespace  — input has trailing ASCII whitespace
#   2. NormalizationDrift  — input != NFC(input)
#
# The remaining four spec sub-threats (EncodingMismatch,
# SignedMessageRule, AuditLogReinterpretation,
# WebhookSignatureDrift) require context the codepoint-only
# detector cannot access; declared in SubThreat but never
# emitted by the v1 detector.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Attribution keys recognised by the conformance harness:
#   stableSize=<N>  — codepoint count of the canonical (NFC + trim) form
#
# Classifications produced by the detector:
#   Clear
#   Hazard:TrailingWhitespace
#   Hazard:NormalizationDrift

# @section Clear
# @level basic

# Empty input is hash-stable (vacuous fixed-point).
; Clear; ; stableSize=0; # empty input

# Pure ASCII "abc" — no NFC change, no trailing whitespace.
61 62 63; Clear; ; stableSize=3; # ASCII "abc"

# Precomposed é (NFC form) — already canonical.
00E9; Clear; ; stableSize=1; # precomposed é

# Han "中文" — CJK with no canonical decomposition.
4E2D 6587; Clear; ; stableSize=2; # 中文

# Mixed ASCII + precomposed accent + internal space.
48 65 6C 6C 6F 20 00E9; Clear; ; stableSize=7; # "Hello é"

# @section Clear
# @level strict

# Internal-only ASCII space between content — only TRAILING
# whitespace is framing.  Pinned to prevent a future refactor
# from accidentally stripping internal whitespace.
61 20 62; Clear; ; stableSize=3; # "a b" internal space

# Trailing U+3000 ideographic space — Unicode whitespace is
# content under this detector's threat model and NOT stripped.
# Pinned to distinguish HashInputStability's ASCII-only trim
# from Bip39Canonical's BIP-39
# {U+0020, U+3000} inventory.
61 3000; Clear; ; stableSize=2; # trailing U+3000 (content not framing)

# @section TrailingWhitespace
# @level basic

# Single trailing U+0020 SPACE.
61 20; Hazard:TrailingWhitespace; 1; stableSize=1; # trailing space

# Trailing U+0009 TAB.
61 09; Hazard:TrailingWhitespace; 1; stableSize=1; # trailing tab

# Trailing U+000A LF.
61 0A; Hazard:TrailingWhitespace; 1; stableSize=1; # trailing LF

# Trailing CRLF — the windows-line-ending shape.  Count = 2,
# both stripped.
61 0D 0A; Hazard:TrailingWhitespace; 1; stableSize=1; # trailing CRLF

# @section NormalizationDrift
# @level basic

# Decomposed é (a + combining acute) — NFC composes to U+00E9.
0065 0301; Hazard:NormalizationDrift; 0; stableSize=1; # decomposed é

# Decomposed á (a + combining acute) — same shape as decomposed é
# but on a different base letter; pins that the detector doesn't
# special-case any specific accent.
0061 0301; Hazard:NormalizationDrift; 0; stableSize=1; # decomposed á

# Decomposed Hangul jamos U+1112 U+1161 U+11AB — NFC composes
# the Choseong + Jungseong + Jongseong triple to the single
# precomposed syllable U+D55C (한).
1112 1161 11AB; Hazard:NormalizationDrift; 0; stableSize=1; # Hangul jamos compose to 한

# Mid-string decomposition: "Hello " + decomposed é + "world".
# Detector reports first divergence position (6, where the
# combining acute sits) and stableSize accounts for the NFC
# collapse.
48 65 6C 6C 6F 20 0065 0301 20 77 6F 72 6C 64; Hazard:NormalizationDrift; 6; stableSize=13; # "Hello é world" mid-string decomposition

# @section TrailingWhitespace
# @level strict

# Combined NFC drift AND trailing whitespace — priority order
# says TrailingWhitespace fires first even though
# NormalizationDrift also applies.  Pins the priority decision
# point in detect.
0065 0301 20; Hazard:TrailingWhitespace; 2; stableSize=1; # decomposed é + trailing space (trailing wins)

# @section EncodingMismatch
# @level basic

# ASCII "abc" labeled with a non-UTF-8 encoding fires
# encodingMismatch.  The codepoint array IS the UTF-8 decode
# by construction; any non-UTF-8 declaration is label drift.
# Context: declaredEnc=utf-16.
61 62 63; Hazard:EncodingMismatch; 0; stableSize=3; declaredEnc=utf-16; # ASCII labeled utf-16

# Same input labeled "latin-1" — fires the same way.
61 62 63; Hazard:EncodingMismatch; 0; stableSize=3; declaredEnc=latin-1; # ASCII labeled latin-1

# @section SignedMessageRule
# @level basic

# RFC 4880 trailing-whitespace rule: input has trailing space
# AND ctx.rfcRule = pgp4880TrailingWhitespace.  The RFC-
# specific verdict outranks the generic trailingWhitespace.
61 20; Hazard:SignedMessageRule; 1; stableSize=1; rfcRule=pgp4880TrailingWhitespace; # trailing space under PGP 4880

# RFC 9580 line-ending rule: input has bare LF (no preceding
# CR) — violates CRLF-only canonicalisation.
61 0A 62; Hazard:SignedMessageRule; 1; stableSize=3; rfcRule=pgp9580LineEnding; # bare LF under PGP 9580

# RFC 8785 NFC requirement: input is decomposed é AND ctx
# selects JSON canonicalisation.  Same condition as
# normalizationDrift, reported under the JSON-specific tag.
0065 0301; Hazard:SignedMessageRule; 0; stableSize=1; rfcRule=rfc8785NfcRequirement; # decomposed é under RFC 8785

# RFC 8259 control-character rule: input has unescaped U+0001
# (Start of Heading) inside a string.  JSON strings must
# escape C0 controls per §7.
61 01 62; Hazard:SignedMessageRule; 1; stableSize=3; rfcRule=rfc8259ControlChar; # unescaped C0 control under RFC 8259

# @section AuditLogReinterpretation
# @level basic

# Audit-log re-read diverges from the as-written form at
# position 2 ("abc" → "abd").  Context carries the original
# written bytes.
61 62 64; Hazard:AuditLogReinterpretation; 2; stableSize=3; asWritten=61 62 63; # written "abc", read "abd"

# Length-mismatch divergence: written is shorter than re-read.
61 62 63 64; Hazard:AuditLogReinterpretation; 3; stableSize=4; asWritten=61 62 63; # written "abc", read "abcd"

# @section WebhookSignatureDrift
# @level basic

# Client bytes "abc" differ from server-recomputed "abd" at
# position 2.  Context carries the server bytes.
61 62 63; Hazard:WebhookSignatureDrift; 2; stableSize=3; serverBytes=61 62 64; # client "abc", server "abd"

# @section EncodingMismatch
# @level strict

# Priority pin: encodingMismatch (priority 1) fires before
# RFC-rule (priority 4) even when both apply.  Input has
# decomposed é AND ctx selects both a non-UTF-8 declared
# encoding AND the RFC 8785 NFC rule.  encodingMismatch wins.
0065 0301; Hazard:EncodingMismatch; 0; stableSize=1; declaredEnc=utf-16; rfcRule=rfc8785NfcRequirement; # priority pin
