# IdentifierFormDriftTest.txt
# UCD version: 16.0.0
# Security-spec version: 0.1.0
# Family: IdentifierFormDrift (UTS #39 + UAX #15)
# License: Apache-2.0
#
# Threat model: an attacker submits an input containing a
# codepoint whose UTS #39 Identifier_Status differs from the
# Identifier_Status of its NFKD-head.  Validators that run
# `isAllowedIdentifier` before normalisation reach a different
# verdict from peers that normalise first — the canonical
# two-stage bypass for usernames, package names, IDN labels,
# and DNS labels.
#
# Detection (single sub-threat in v1, direction-agnostic):
#   IdentifierStatusShift (basePos, cp) — first input position
#     whose `isAllowedStatus` differs from the NFKD-head's
#     `isAllowedStatus`.
#
# Distinct from neighbouring detectors:
#   HomoglyphConfusable     — single-form skeleton matching
#   MixedScriptAdmissibility — single-form script set
#   WidthClassConfusion     — EAW class change under NFKD
#   NfcIdempotenceWitness    — any NFC/NFKC change
# IdentifierFormDrift fires on the *identifier-verdict
# transition* itself, which the single-form identity-spoofing
# detectors miss by construction and which the form-stability
# detectors don't translate into an identity claim.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the detector:
#   Clear (= identifier status is form-stable)
#   Hazard:IdentifierStatusShift

# @section Clear
# @level basic

# ASCII Hello — every cp Allowed with identity NFKD.
0048 0065 006C 006C 006F; Clear; ; shift_count=0; # Hello

# Plain ASCII admin — the canonical credential.
0061 0064 006D 0069 006E; Clear; ; shift_count=0; # admin

# Plain Han 中文 — both cps Allowed, identity NFKD.
4E2D 6587; Clear; ; shift_count=0; # 中文

# Greek lowercase αβγ — all Allowed, identity NFKD.
03B1 03B2 03B3; Clear; ; shift_count=0; # αβγ

# Note: precomposed Hangul syllables (e.g. U+D55C 한) are
# Allowed but their NFKD-head jamos (e.g. U+1112) are Restricted
# in UTS #39.  That shift runs Allowed -> Restricted, which a
# validator cannot be walked through: it accepts the input in
# both forms.  The detector reports Restricted -> Allowed only,
# so pure Hangul is clear; the U+D55C row below carries that
# verdict beside the shifts it is contrasted with.

# Cyrillic привет — every cp Allowed.
041F 0440 0438 0432 0435 0442; Clear; ; shift_count=0; # привет (Russian)

# Arabic word — all Allowed, identity NFKD.
0641 0644 062F; Clear; ; shift_count=0; # Arabic word

# Hiragana word — Allowed, identity NFKD.
3042 3044 3046; Clear; ; shift_count=0; # あいう

# Mixed ASCII identifier with underscore.
0066 006F 006F 005F 0062 0061 0072; Clear; ; shift_count=0; # foo_bar

# @section IdentifierStatusShift
# @level basic

# Mathematical Italic Small A (U+1D44E) — Restricted (math
# alphanumeric block).  NFKD-head = U+0061 'a' (Allowed).  The
# canonical IdentifierFormDrift attack: bypass an
# `isAllowedIdentifier` check by
# routing through a normalising stage.
1D44E; Hazard:IdentifierStatusShift; 0; shift_count=1; # 𝑎 (math italic a)

# Mathematical Italic "admin" — five-cp math identifier whose
# NFKD folds to "admin".  Shift fires at position 0.
1D44E 1D451 1D45A 1D456 1D45B; Hazard:IdentifierStatusShift; 0; shift_count=5; # 𝑎𝑑𝑚𝑖𝑛

# Fullwidth Ａ (U+FF21) — Restricted compatibility form, NFKD-head
# U+0041 'A' Allowed.  Same threat shape as the Math Italic case
# but via the F-class fold.
FF21; Hazard:IdentifierStatusShift; 0; shift_count=1; # Ａ

# Circled A (U+24B6) — Restricted (Default_Ignorable / enclosed
# alphanumeric).  NFKD-head U+0041 Allowed.
24B6; Hazard:IdentifierStatusShift; 0; shift_count=1; # Ⓐ

# ﬁ ligature (U+FB01) — Restricted compatibility form, NFKD-head
# U+0066 'f' Allowed.  EAW=N so WidthClassConfusion misses this;
# NfcIdempotenceWitness catches the form change but not the
# identity claim shift; IdentifierFormDrift is the identity-side
# detector for this codepoint class.
FB01; Hazard:IdentifierStatusShift; 0; shift_count=1; # ﬁ

# Roman numeral Ⅳ (U+2163) — Restricted, NFKD-head U+0049 'I'
# Allowed.  Letter-like-symbols-as-identifier attack.
2163; Hazard:IdentifierStatusShift; 0; shift_count=1; # Ⅳ

# ASCII prefix + math italic letter at position 3 — shift at
# position 3, not 0.
0061 0062 0063 1D44E; Hazard:IdentifierStatusShift; 3; shift_count=1; # abc𝑎

# Precomposed Korean syllable 한 (U+D55C, Allowed) whose NFKD-head
# is U+1112 HANGUL CHOSEONG HIEUH (Restricted in UTS #39
# IdentifierStatus.txt).  The shift runs Allowed -> Restricted,
# the direction that gains an attacker nothing: a validator that
# accepts the input before normalisation also accepts it after.
# The detector reports the opposite direction only, so pure
# Korean text is clear.
D55C; Clear; ; shift_count=0; # Korean precomposed → jamo Restricted, harmless direction

# Math Bold Capital A (U+1D400) — Restricted, NFKD-head U+0041.
1D400; Hazard:IdentifierStatusShift; 0; shift_count=1; # math bold A

# Math Sans-Serif lowercase 'a' (U+1D5BA) — Restricted, NFKD-head U+0061.
1D5BA; Hazard:IdentifierStatusShift; 0; shift_count=1; # math sans-serif a

# Math Fraktur Capital A (U+1D504) — Restricted, NFKD-head U+0041.
1D504; Hazard:IdentifierStatusShift; 0; shift_count=1; # math fraktur A

# Roman numeral C (U+216D) — Restricted, NFKD-head U+0043 Allowed.
216D; Hazard:IdentifierStatusShift; 0; shift_count=1; # roman C

# Circled digit 1 (U+2460) — Restricted, NFKD-head U+0031 Allowed.
2460; Hazard:IdentifierStatusShift; 0; shift_count=1; # circled 1

# Fullwidth lowercase ｐａｓｓ — every cp Restricted, NFKD-head
# Allowed.  Password-validator bypass shape.
FF50 FF41 FF53 FF53; Hazard:IdentifierStatusShift; 0; shift_count=4; # fullwidth pass

# Fullwidth digits — every cp shifts.
FF10 FF11 FF12 FF13; Hazard:IdentifierStatusShift; 0; shift_count=4; # fullwidth 0123

# ASCII "user" with fullwidth digit at position 4 — single shift.
0075 0073 0065 0072 FF11; Hazard:IdentifierStatusShift; 4; shift_count=1; # user + fullwidth 1

# Math Italic "password" — eight-cp math identifier whose NFKD
# folds to "password".
1D45D 1D44E 1D460 1D460 1D464 1D45C 1D45F 1D451; Hazard:IdentifierStatusShift; 0; shift_count=8; # math italic password
