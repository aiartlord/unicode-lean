# SourceDisplayDivergenceTokenizedTest.txt
# UCD version: 16.0.0
# Security-spec version: 0.1.0
# Family: D1 (v1.5 region-aware tokenization)
# Grammar: Rust (`Language.rust`)
# License: Apache-2.0
#
# Companion fixture to SourceDisplayDivergenceTest.txt that
# exercises D1's v1.5 region-aware behaviour.  Every row is
# evaluated with `detect input .rust`, so the verdicts here may
# differ from the v1 fixture's verdicts on the same byte sequence
# whenever a sub-detector's hit lands inside a string literal,
# line comment, or block comment.
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the D1 detector are unchanged from
# the v1 fixture — only the filtering changes.

# @section Clear
# @level basic

# ASCII Hello — no payload, all code.
0048 0065 006C 006C 006F; Clear; ; # Hello

# VS payload INSIDE a Rust string literal.  Under .none the C2
# hit would fire D1; under .rust the position is inside the
# string region, so the hit is filtered out.
0022 0041 FE00 0022; Clear; ; # "A + VS1" — VS in string

# Bidi format-control INSIDE a Rust line comment.  Under .none
# the C5 hit would fire D1; under .rust the comment runs to
# end-of-input and the RLO sits inside it.
002F 002F 202E; Clear; ; # // + RLO

# Tag-block character INSIDE a Rust block comment.
002F 002A E0001 002A 002F; Clear; ; # /* + TAG + */

# VS payload INSIDE a Rust raw string `r#"..."#`.
0072 0023 0022 0041 FE00 0022 0023; Clear; ; # r#"A+VS1"#

# Latin "react" identifier in a normal Rust string literal — no
# I1 confusable hit because all letters are ASCII.
0022 0072 0065 0061 0063 0074 0022; Clear; ; # react in string

# Math-Italic 'a' INSIDE a Rust string literal — I1.MathAlpha
# reports positions=[1] which is inside the string region.  Hit
# is filtered out by D1's region-aware tokenization.
#
# (Note: I1.TargetMatch fires whole-string with positions=[], so
# wrapping a Cyrillic-letter typosquat in a string does NOT
# filter the hit — see the Nethereum-in-string row in the Hazard
# section below.  Whole-string verdicts intentionally bypass
# region filtering since the hazard is the whole identifier, not
# any particular codepoint.)
0022 1D44E 0022; Clear; ; # math-italic a in string

# Bidi control INSIDE a Rust block comment — also filtered.
002F 002A 202E 002A 002F; Clear; ; # block-comment RLO

# Multiple hazard codepoints (RLO + VS + ZWJ) all INSIDE a
# string literal — all filtered.
0022 202E 0041 FE00 200B 0022; Clear; ; # multiple hazards in string

# @section Hazard
# @level basic

# RLO BEFORE a Rust string literal — RLO sits in the leading
# code region, so the C5 hit is in code and D1 fires.
202E 0022 0041 0022; Hazard:BidiControl; ; # RLO + A in code

# VS attached to a letter immediately AFTER the closing quote —
# the VS position is in code (the string region ends at the
# closing quote).
0022 0041 0022 FE00; Hazard:VariationSelector; ; # A in string + VS1 after

# Tag-block character BEFORE a Rust block comment.
E0001 002F 002A 002A 002F; Hazard:TagBlock; ; # TAG + block comment

# Pure code: VS attached to letter, no string or comment
# anywhere — same verdict as v1.
0041 FE00; Hazard:VariationSelector; ; # A + VS1 in code

# Zero-width joiner outside any RGI ZWJ sequence, in code.
0041 200D 0042; Hazard:ZeroWidth; ; # A ZWJ B in code

# Cyrillic Nethereum typosquat in code (no string wrapper) —
# I1 hit is in code and fires D1.
004E 0065 0074 0068 0065 0072 0435 0075 006D; Hazard:IdentifierHomoglyph; ; # Cyrillic-е Nethereum in code

# RLO + Cyrillic identifier — compound in code.
202E 004E 0065 0074 0068 0065 0072 0435 0075 006D; Hazard:Compound; ; # RLO + Cyrillic Nethereum

# Code AFTER a string boundary — VS hit in trailing code segment.
0022 0048 0069 0022 0020 0041 FE00; Hazard:VariationSelector; ; # "Hi" + space + A + VS1

# Multi-region: code then string then code containing a hazard
# at the very end.
0066 0028 0022 0078 0022 0029 202E; Hazard:BidiControl; ; # f("x") + trailing RLO

# Tag block straddling a string boundary — appears AFTER the
# closing quote so the hit is in code.
0022 0048 0069 0022 E0041; Hazard:TagBlock; ; # Hi + trailing tag-A

# Cyrillic Nethereum typosquat INSIDE a string literal.  I1's
# TargetMatch reports positions=#[] (whole-string verdict), so
# D1's region filter passes the hit through unconditionally.  A
# string-wrapper does NOT hide a registered-target typosquat
# under v1.5 tokenization — by design, since the typosquat is a
# property of the identifier, not its enclosing source region.
0022 004E 0065 0074 0068 0065 0072 0435 0075 006D 0022; Hazard:IdentifierHomoglyph; ; # Cyrillic-е Nethereum in string (TargetMatch bypass)
