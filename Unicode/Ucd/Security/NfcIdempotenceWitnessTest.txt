# Family: NfcIdempotenceWitness (UAX #15)

#
# Threat model: an attacker submits text in a non-canonical form
# that passes one stage of validation (which doesn't normalise)
# but matches a stored canonical value at another stage (which
# does).  The two-system bypass shape applies to any field that
# is compared after being normalised on one side but not the
# other — usernames, file paths, content hashes, cache keys.
#
# Detection (priority order, both reachable):
#   1. NonNfcForm        — input != NFC(input).  The input is
#                          not in canonical composition; combining
#                          marks are decomposed or wrongly
#                          ordered.
#   2. NonNfkcCompatForm — input == NFC(input) but input
#                          != NFKC(input).  Compatibility-only
#                          form (ligatures, etc.).  Reached
#                          when NonNfcForm doesn't fire first.
#
# Distinct from neighbouring form-stability detectors:
#   CaseExpansionMismatch — length change under case folding
#   WidthClassConfusion   — EAW class change under NFKD
# NfcIdempotenceWitness is the residual catch-all for shapes
# that change under NFC or NFKC but don't fit the more specific
# form-stability detectors.  In particular it fires on U+FB01 ﬁ
# (EAW = N, missed by WidthClassConfusion).
#
# Format:
#   <hex codepoints>; <classification>; <positions>; <attribution>; # <name>
#
# Classifications produced by the detector:
#   Clear
#   Hazard:NonNfcForm
#   Hazard:NonNfkcCompatForm

# @section Clear
# @level basic

# ASCII Hello — in NFC and NFKC.
0048 0065 006C 006C 006F; Clear; ; nfc_len=5; nfkc_len=5; # Hello

# Pre-composed é (U+00E9) — canonical NFC = #[00E9] = input.
00E9; Clear; ; nfc_len=1; nfkc_len=1; # é precomposed

# Plain Han 中文 — identity under NFC/NFKC.
4E2D 6587; Clear; ; nfc_len=2; nfkc_len=2; # 中文

# Korean Hangul 한 — canonical NFC composition.
D55C; Clear; ; nfc_len=1; nfkc_len=1; # 한 (precomposed)

# Cyrillic привет — identity under NFC/NFKC.
041F 0440 0438 0432 0435 0442; Clear; ; nfc_len=6; nfkc_len=6; # привет

# Greek lowercase αβγ — identity under both forms.
03B1 03B2 03B3; Clear; ; nfc_len=3; nfkc_len=3; # αβγ

# Arabic letters — identity under both forms.
0627 0628 0629; Clear; ; nfc_len=3; nfkc_len=3; # ابة

# Plain ASCII URL — identity under both forms.
0068 0074 0074 0070 003A 002F 002F; Clear; ; nfc_len=7; nfkc_len=7; # http://

# @section NonNfcForm
# @level basic

# e + combining acute (decomposed é) — NFC folds to #[00E9], so
# input differs from NFC at position 0.  The canonical
# decomposed-vs-precomposed mismatch.
0065 0301; Hazard:NonNfcForm; 0; nfc_len=1; nfkc_len=1; # e + combining acute → é

# Two decomposed combining vowels — a + combining acute, e +
# combining grave.  First decomp at position 0.
0061 0301 0065 0300; Hazard:NonNfcForm; 0; nfc_len=2; nfkc_len=2; # á + è decomposed

# Decomposed o + combining diaeresis — NFC = #[00F6].
006F 0308; Hazard:NonNfcForm; 0; nfc_len=1; nfkc_len=1; # o + diaeresis → ö

# Decomposed Hangul jamos — toNFC composes them back to D55C.
1112 1161 11AB; Hazard:NonNfcForm; 0; nfc_len=1; nfkc_len=1; # Hangul jamos compose to D55C

# u + combining diaeresis — NFC = precomposed u-with-diaeresis.
0075 0308; Hazard:NonNfcForm; 0; nfc_len=1; nfkc_len=1; # u + diaeresis decomposed

# n + combining tilde — NFC = precomposed n-with-tilde.
006E 0303; Hazard:NonNfcForm; 0; nfc_len=1; nfkc_len=1; # n + tilde decomposed

# c + combining cedilla — NFC = precomposed c-with-cedilla.
0063 0327; Hazard:NonNfcForm; 0; nfc_len=1; nfkc_len=1; # c + cedilla decomposed

# Long Spanish word with the n-with-tilde decomposed at position 2.
006E 0069 006E 0303 006F; Hazard:NonNfcForm; 2; nfc_len=4; nfkc_len=4; # nino-tilde decomposed at pos 2

# o + combining circumflex — NFC = precomposed o-with-circumflex.
006F 0302; Hazard:NonNfcForm; 0; nfc_len=1; nfkc_len=1; # 한 jamos → 한

# @section NonNfkcCompatForm
# @level basic

# ﬁ ligature (U+FB01) — canonical NFC = #[FB01] = input, but
# NFKC = #[0066, 0069] differs.  EAW = N so WidthClassConfusion
# doesn't see this as a width-class fold; NfcIdempotenceWitness
# is the right detector.
FB01; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=2; # ﬁ → fi compat

# ﬃ ligature (U+FB03) — NFC stable, NFKC = #[0066, 0066, 0069].
FB03; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=3; # ﬃ → ffi compat

# Roman numeral Ⅳ (U+2163) — NFC stable, NFKC = #[0049, 0056] "IV".
2163; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=2; # Ⅳ → IV compat

# Circled digit ① — NFKC = "1".
2460; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=1; # circled-1 to ASCII 1 compat

# Superscript ² — NFKC = "2".
00B2; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=1; # superscript 2 compat

# Vulgar fraction ½ — NFKC = "1⁄2" three cps.
00BD; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=3; # vulgar half compat

# Arabic ligature SALLALLAHOU ALAYHE WASALLAM (U+FDFA) — NFC stable,
# NFKC = 18 cps.  The same codepoint also fires NormalizationBomb;
# this fixture pins the residual-form-stability finding.
FDFA; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=18; # FDFA Arabic 18-cp expansion

# Square HV (U+33CD) — NFKC = "HV".
33CD; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=2; # square HV compat

# Halfwidth katakana 'ｱ' (FF71) — NFKC = the wide form U+30A2.
FF71; Hazard:NonNfkcCompatForm; 0; nfc_len=1; nfkc_len=1; # halfwidth katakana → wide compat
